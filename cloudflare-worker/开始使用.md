### 通过命令行管理凭证

\`\`\`bash
# 查看所有凭证
npx wrangler kv key list --binding=CREDENTIALS_KV --prefix=cred:

# 添加凭证
npx wrangler kv key put --binding=CREDENTIALS_KV "cred:test_1234567890" '{
  "bearer_token": "你的Token",
  "user_id": "你的用户ID",
  "created_at": 1234567890,
  "expires_in": 2592000
}'

# 删除凭证
npx wrangler kv key delete --binding=CREDENTIALS_KV "cred:test_1234567890"
